import { readFile, writeFile, mkdir } from 'node:fs/promises';
import { resolve, join } from 'node:path';
import { pathToFileURL } from 'node:url';
import { chromium } from 'playwright';
import { runAcceptance } from '../../src/acceptance/runner.mjs';

const output = resolve('core/workspace/_runs/acceptance-dynamic-phase-20261017');
await mkdir(output, { recursive: true });
const source = await readFile('core/test/acceptance-runner.test.mjs', 'utf8');
const end = source.indexOf("test('acceptance runner reproduces");
if (end < 0 || !source.includes("['red','green','blue'][i++%3]},33)")) throw new Error('Original periodic fixture not found');
// Reuse the actual pre-fix fixture and plan helper without registering tests.
const extracted = source.slice(0, end).replaceAll("from '../src/", "from '../../src/") + '\nexport { fixture, planFor };\n';
const modulePath = resolve('core/workspace/_runs/acceptance-dynamic-phase-original-fixture.probe.mjs');
await writeFile(modulePath, extracted);
await writeFile(join(output, 'original-fixture-prefix.mjs.txt'), source.slice(0, end));
const { fixture, planFor } = await import(pathToFileURL(modulePath).href);
const cleanups = [], api = await fixture({ after(fn) { cleanups.push(fn); } });
const launch = chromium.launch.bind(chromium), results = [];
try {
  for (const [name, repeats] of [['same-phase', 3], ['different-phase', 1]]) {
    const captures = [];
    chromium.launch = async options => {
      const browser = await launch(options), newContext = browser.newContext.bind(browser);
      browser.newContext = async options => {
        const context = await newContext(options);
        await context.addInitScript(() => {
          const original = window.setInterval.bind(window);
          window.setInterval = (callback, ms, ...args) => {
            if (ms === 33) { window.__noiseTick = callback; return 1; }
            return original(callback, ms, ...args);
          };
        });
        const newPage = context.newPage.bind(context);
        context.newPage = async () => {
          const page = await newPage(), screenshot = page.screenshot.bind(page);
          let capture = 0;
          page.screenshot = async options => {
            const ticks = capture++ ? repeats : 1;
            const background = await page.evaluate(ticks => {
              for (let index = 0; index < ticks; index++) window.__noiseTick();
              return getComputedStyle(document.querySelector('#dynamic')).backgroundColor;
            }, ticks);
            captures.push({ url: page.url(), capture, ticks, background });
            return screenshot(options);
          };
          return page;
        };
        return context;
      };
      return browser;
    };
    const report = await runAcceptance(planFor(api.origin, { routes: ['/dynamic'] }), { runId: name, outputDir: join(output, name) });
    const visual = report.checks.find(check => check.dimension === 'visual');
    results.push({ name, originalIntervalMs: 33, cycleTicks: 3, cycleMs: 99, captures, visual });
    await writeFile(join(output, name + '-report.json'), JSON.stringify(report, null, 2));
  }
  await writeFile(join(output, 'results.json'), JSON.stringify(results, null, 2));
  console.log(JSON.stringify(results.map(({name,captures,visual}) => ({name,captures,status:visual.status,verdict:visual.verdict,
    score:visual.actual.score,baselineStability:visual.actual.baselineStability,targetStability:visual.actual.targetStability})),null,2));
} finally {
  chromium.launch = launch;
  for (const cleanup of cleanups.reverse()) await cleanup();
}
